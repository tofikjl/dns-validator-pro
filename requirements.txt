def test_dns_server(server, test_domain="google.com", timeout=3, attempts=3):

"Test a standard DNS server (UDP). Returns (success rate, average time)"

resolver = dns.resolver.Resolver() 

resolver.nameservers = [server] 
resolver. timeout = timeout 
resolver.lifetime = timeout 
success = 0 
total_time = 0 
for_in range(attempts): 
try: 
start = time.time() 
resolver.resolve(test_domain, 'A') 
elapsed = (time.time() - start) * 1000 
total_time += elapsed 
success += 1 
except Exception: 
pass. pass 
success_rate = (success / attempts) * 100 
avg_time = total_time / success if success > 0 else float('inf') 
return (server, success_rate, avg_time)

def export_results(results, filename, format='json'): 
“Export results to JSON or CSV file” 
if format == 'json': 
with open(filename, 'w') as f: 
json.dump(results, f, indent=4) 
elif format == 'csv': 
with open(filename, 'w', newline='') as f: 
writer = csv.writer(f) 
writer.writerow(['Server', 'Success Rate (%)', 'Avg Response Time (ms)']) 
for r in results: 
writer.writerow([r[0], r[1], r[2] if r[2] != float('inf') else 'N/A']) 
print(f"[✓] Results exported to {filename}")

def main(): 
parser = argparse.ArgumentParser(description="DNS Validator Pro - Find the fastest and most reliable DNS server.") 
parser.add_argument("--servers", nargs="+", help="List of custom DNS servers (IPs)") 
parser.add_argument("--doh", action="store_true", help="Include DNS over HTTPS (DoH) servers in test") 
parser.add_argument("--doh-only", action="store_true", help="Test only DoH servers (ignores --servers and default UDP servers)")

parser.add_argument("--export", help="Export results to file (specify filename). Format determined by extension (.json or .csv)")
